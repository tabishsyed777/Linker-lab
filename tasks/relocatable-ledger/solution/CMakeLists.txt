cmake_minimum_required(VERSION 3.20)
project(ledgercalc C)
add_library(money SHARED money.c)
add_library(ledger SHARED ledger.c)
target_link_libraries(ledger PRIVATE money)
add_executable(ledgercalc main.c)
target_link_libraries(ledgercalc PRIVATE ledger)
set_target_properties(ledgercalc PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
set_target_properties(ledger PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS ledgercalc ledger money RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
